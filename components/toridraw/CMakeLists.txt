# ToriDraw, from oldschool-clientc/3rd/toridraw.
#
# ONE TRANSLATION UNIT. toridraw_unity.c is the library's own integration
# surface -- "add -I<dir>/3rd/toridraw and compile this file", and nothing
# else -- so this component names one source instead of the dozen the old
# src2 layout made us track by hand. A file added to the library upstream
# arrives here without an edit, which is the point of a unity build and was
# the standing maintenance cost of the list it replaces.
#
# The linker drops what this client never calls: IDF builds with
# -ffunction-sections -fdata-sections and links with --gc-sections, so the
# scene graph, the font, the sprite and the 2D layer cost nothing in the image
# despite being compiled.

if(NOT DEFINED CLIENTC_DIR)
    get_filename_component(CLIENTC_DIR
        "${CMAKE_CURRENT_LIST_DIR}/../../3rd/oldschool-clientc" ABSOLUTE)
endif()
file(TO_CMAKE_PATH "${CLIENTC_DIR}" CLIENTC_DIR)

set(TORIDRAW_DIR "${CLIENTC_DIR}/3rd/toridraw")
set(TORIDRAW_TABLES_C "${CMAKE_BINARY_DIR}/gen/toridraw_tables_precomputed.c")

if(NOT EXISTS "${TORIDRAW_DIR}/toridraw_unity.c")
    message(FATAL_ERROR
        "toridraw not found at ${TORIDRAW_DIR}. "
        "Set -DCLIENTC_DIR=<path to oldschool-clientc>.")
endif()

# The hand-written flat presorted-run kernel. It is the reason this client can
# afford a painter at all -- see impl/raster/asm/tri.flat.rgb565.xtensa.S and
# 3rd/toridraw/test/xtensa for what it is scored and measured against.
set(TORIDRAW_ASM
    "${TORIDRAW_DIR}/impl/raster/asm/tri.flat.rgb565.xtensa.S"
    "${TORIDRAW_DIR}/impl/raster/asm/tri.gouraudhsllightness.rgb565.xtensa.S")

idf_component_register(
    SRCS
        "${TORIDRAW_DIR}/toridraw_unity.c"
        ${TORIDRAW_ASM}
    INCLUDE_DIRS
        "${TORIDRAW_DIR}"
)

#
# The table unit is added AFTER registration, not in SRCS.
#
# It does not exist until the root CMakeLists runs its generator, and CMake
# checks that every listed source exists at configure time -- but the obvious
# fix, set_source_files_properties(... GENERATED) before the register, cannot
# go there: ESP-IDF evaluates this file in SCRIPT mode first to collect
# component requirements, and that command is not scriptable. Everything below
# the register runs only in the real configure pass, where both work.
#
set_source_files_properties("${TORIDRAW_TABLES_C}" PROPERTIES GENERATED TRUE)
target_sources(${COMPONENT_LIB} PRIVATE "${TORIDRAW_TABLES_C}")

# PUBLIC because toripixel_t is in this project's face: main.c writes the
# framebuffer these kernels fill, and a caller compiled for a different format
# would disagree with the library about how wide a pixel is. The generated
# table unit _Static_asserts the same thing, so a mismatch is a compile error
# rather than a screen of wrong colours.
target_compile_definitions(${COMPONENT_LIB} PUBLIC
    TORIDRAW_PIXEL_FORMAT=TORIDRAW_PF_RGB565_BE
)

target_compile_definitions(${COMPONENT_LIB} PRIVATE
    # The palette and the trigonometric tables as `const`, so they land in
    # flash instead of costing 185 KB of a 512 KB part. Generated by
    # 3rd/toridraw/tools/toridraw_tables_gen.c -- see scripts/gen-const-tables.ps1.
    TORIDRAW_TABLES_PRECOMPUTED
    # Arms the Xtensa arm of graphics/raster/flat/flat_tri_asm.h. Without it
    # the header declines and the .S above is dead bytes; with it and the
    # wrong pixel format the header ALSO declines, which is the gate doing its
    # job rather than a misconfiguration to chase.
    TORIDRAW_FLAT_TRI_XTENSA_ASM=1
    TORIDRAW_GOURAUD_TRI_XTENSA_ASM=1
    # The gouraud colour gradients divide exactly instead of taking a double
    # reciprocal this core cannot afford. The C must agree, or the two halves
    # of the raster disagree by a shade step; see the kernel header.
    TORIDRAW_GOURAUD_STEP_IDIV=1
    # The panel's byte order, so the .S picks the matching entry points.
    TORIDRAW_XTENSA_RGB565_BE=1
)

target_compile_options(${COMPONENT_LIB} PRIVATE
    -Wno-unused-function
    -Wno-unused-variable
    -Wno-unused-but-set-variable
    -Wno-unused-parameter
)

# -mtext-section-literals puts the kernel's literal pool in .text beside the
# l32r that reads it; without it the literals land in a section the app linker
# script does not place in range and the link fails on l32r relocation.
set_source_files_properties(${TORIDRAW_ASM}
    PROPERTIES COMPILE_OPTIONS "-mtext-section-literals;-DTORIDRAW_XTENSA_RGB565_BE=1")

target_link_libraries(${COMPONENT_LIB} PRIVATE m)
